        async function ensureUserProfile(authUser, extraMeta = {}) {
            if (!authUser) return null;
            const email = authUser.email || extraMeta.email || "";
            const shouldBeAdmin = email.toLowerCase() === ADMIN_EMAIL.toLowerCase();

            try {
                const { data: existing, error: fetchErr } = await supabaseClient
                    .from("profiles")
                    .select("*")
                    .eq("id", authUser.id)
                    .maybeSingle();

                if (fetchErr) throw fetchErr;

                if (existing) {
                    const adminFlag = Boolean(existing.is_admin) || shouldBeAdmin;
                    if (adminFlag !== Boolean(existing.is_admin)) {
                        await supabaseClient.from("profiles").update({ is_admin: adminFlag }).eq("id", authUser.id);
                        existing.is_admin = adminFlag;
                    }
                    state.profile = existing;
                    return existing;
                }

                const newProfile = {
                    id: authUser.id,
                    email: email,
                    team_name: extraMeta.team_name || authUser.user_metadata?.team_name || email.split("@")[0] || "MLBB TEAM",
                    mlbb_id: extraMeta.mlbb_id || authUser.user_metadata?.mlbb_id || "00000000 (0000)",
                    balance: 0,
                    is_admin: shouldBeAdmin
                };

                const { data: inserted, error: insertErr } = await supabaseClient
                    .from("profiles")
                    .insert([newProfile])
                    .select()
                    .single();

                if (insertErr) throw insertErr;

                state.profile = inserted;
                return inserted;
            } catch (err) {
                console.warn("Profile fallback triggered:", err);
                activateFallbackBanner("ไม่สามารถเข้าถึงตาราง profiles บน Supabase ได้ชั่วคราว (อาจติด RLS Policy) เปิดโหมด Preview ชั่วคราวให้ใช้งาน");
                state.profile = {
                    id: authUser.id,
                    email: email,
                    team_name: extraMeta.team_name || authUser.user_metadata?.team_name || "LANXANG PRO",
                    mlbb_id: extraMeta.mlbb_id || authUser.user_metadata?.mlbb_id || "88492014 (2091)",
                    balance: 500000,
                    is_admin: shouldBeAdmin
                };
                return state.profile;
            }
        }
